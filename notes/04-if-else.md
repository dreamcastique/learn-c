# C - Урок 04:

## Пройдено
- работа `if`, `else if`, `else`
- использование `switch`, `case`, `break`, `default`
- `switch` для проверки одного значения, выбираем `switch`, если кол-во проверок больше 3-х
- код возврата это int, возвращаемый во время исполнения программы: 0 - успешно, НЕ 0 - ожидается ошибка

## Код
```c
#include <stdio.h>  // подключаем ввод/вывод

int main(void) {  // главная функция
    int age;      // возраст зрителя
    int place;    // выбранное место
    int price;    // итоговая цена

    printf("Введите возраст: ");  // просим возраст
    scanf("%d", &age);            // читаем возраст

    // Определяем базовую цену по возрасту
    if (age < 0) {                         // если возраст отрицательный
        printf("Возраст не может быть отрицательным.\n");
        return 1;                          // завершаем программу с ошибкой
    } else if (age < 7) {                  // если возраст меньше 7
        price = 0;                         // бесплатно
    } else if (age < 18) {                 // если возраст от 7 до 17
        price = 150;                       // детская цена
    } else {                               // во всех остальных случаях
        price = 300;                       // взрослая цена
    }

    printf("Выберите место (1 - балкон, 2 - партер, 3 - ложа): ");
    scanf("%d", &place);                   // читаем номер места

    // Добавляем надбавку за место через switch
    switch (place) {                       // проверяем значение place
        case 1:                            // если place равно 1
            printf("Выбрано: балкон\n");
            break;                         // выходим из switch
        case 2:                            // если place равно 2
            price = price + 100;           // надбавка 100
            printf("Выбрано: партер\n");
            break;
        case 3:                            // если place равно 3
            price = price + 250;           // надбавка 250
            printf("Выбрано: ложа\n");
            break;
        default:                           // если ни один case не подошёл
            printf("Неизвестное место, надбавка не применена.\n");
            break;
    }

    printf("Итоговая цена: %d руб.\n", price);  // выводим цену
    return 0;                                   // успешное завершение
}
```
