# C - Урок 10:

## Пройдено
- `struct` - это составной тип (объединяет несколько полей), нужна для хранения нескольких значений разных типов
- `struct` - это отдельная инструкция, поэтому после } нужна ;.


## Код
```c
#include <stdio.h>

// Описываем структуру Student.
// Это шаблон: какие данные будут у одного студента.
struct Student {
    char name[50];   // поле name — строка до 49 символов + '\0'
    int age;         // поле age — целое число
    double average;  // поле average — дробное число
};                   // здесь обязательно нужна точка с запятой

int main() {
    // Создаём одну структуру и сразу заполняем поля.
    struct Student s1 = {"Ivan", 19, 4.5};

    // Можно использовать назначенные инициализаторы (C99+).
    struct Student s2 = {.name = "Maria", .age = 20, .average = 4.8};

    // Массив структур: три студента.
    struct Student group[3] = {
        {"Petr", 18, 4.1},
        {"Anna", 19, 4.7},
        {"Oleg", 20, 3.9}
    };

    // Печатаем первого студента.
    printf("Студент 1: %s, возраст %d, средний балл %.1f\n",
           s1.name, s1.age, s1.average);

    // Печатаем второго студента.
    printf("Студент 2: %s, возраст %d, средний балл %.1f\n",
           s2.name, s2.age, s2.average);

    // Проходим по массиву структур.
    printf("\nГруппа:\n");
    for (int i = 0; i < 3; i++) {
        printf("%s — %d лет, балл %.1f\n",
               group[i].name, group[i].age, group[i].average);
    }

    // Указатель на структуру.
    struct Student *p = &s1;

    // Через указатель меняем поле age.
    p->age = 21;

    // Проверяем, что изменилось именно поле s1.age.
    printf("\nПосле изменения через указатель: %s — %d лет\n",
           s1.name, s1.age);

    return 0;
}
```
